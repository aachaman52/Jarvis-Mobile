cmake_minimum_required(VERSION 3.22.1)
project(jarvis_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
include(FetchContent)
# Fixed release, not moving master. Requires network only for the first build.
FetchContent_Declare(llama_cpp GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git GIT_TAG b5046 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(llama_cpp)
add_library(jarvis_llama SHARED jarvis_llama.cpp)
target_link_libraries(jarvis_llama PRIVATE llama android log)
target_link_options(jarvis_llama PRIVATE "-Wl,-z,max-page-size=16384")
